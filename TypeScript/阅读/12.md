// Parameters 获取函数的参数类型

  

type ParametersRes = Parameters<(name: string, age: number)=>{}>

  

// temp

type ParametersTemp<Func extends Function> =

Func extends (...args: infer p) => unknown

? p

: never

  

type ParametersRes2 = ParametersTemp<(name: string, age: number)=>{}>

  
  

// ReturnType 获取函数返回值的类型

  

type ReturnTypeRes = ReturnType<()=>string>

  
  

// temp

type ReturnTypeTemp<Func extends Function> =

Func extends () => infer P

? P

: never

  
  

type ReturnTypeRes2 = ReturnTypeTemp<()=>string>

  
  

// ConstructorParameters 提取构造器的参数类型

  

interface PersonConstructor {

new(name: string): Person;

}

  

class Person {

name: string;

constructor(name: string) {

this.name = name;

}

}

  

type ConstructorParametersRes = ConstructorParameters<PersonConstructor>

  

// temp

type ConstructorParametersTemp<T extends abstract new(...args: any) => any> =

T extends abstract new(...args: infer P) => any

? P

: never

  
  

type ConstructorParametersRes2 = ConstructorParametersTemp<PersonConstructor>

  
  
  
  

// InstanceType 返回构造器返回值类型

  

type InstanceTypeRes = InstanceType<PersonConstructor>

  

// temp

type InstanceTypeTemp<T extends abstract new(...args: any) => any> =

T extends abstract new(...args: any) => infer P

? P

: never

  
  

type InstanceTypeRes2 = InstanceTypeTemp<PersonConstructor>

  
  
  

// ThisParameterType 获取函数this 类型

  

function hello(this: Person){

console.log(this.name)

}

  

type ThisParameterTypeRes = ThisParameterType<typeof hello>

// type ThisParameterTypeRes = ThisParameterType<(this: Person) => void>

  

// temp

  

type ThisParameterTypeTemp<T> =

T extends (this: infer U, ...args: any[]) => any

? U

: unknown

  

type ThisParameterTypeRes2 = ThisParameterTypeTemp<typeof hello>

  
  

// OmitThisParameter 去除函数中指定的this类型

  

function say(this: Person, age: number){

console.log(this.name)

return this.name + ' ' + age

}

  

type OmitThisParameterRes = OmitThisParameter<typeof say>

  

// temp

type OmitThisParameterTemp<T> =

unknown extends ThisParameterType<T>

? T

: T extends (...args: infer A) => infer B

? (...args: A) => B

: T

  

type OmitThisParameterRes2 = OmitThisParameterTemp<typeof say>

  
  

// Partial 把对象中的属性都改为可选 key?

  

type PartialRes = Partial<{name: string, age: number}>

  

type PartialTemp<T> = {

[K in keyof T]?: T[K]

}

  

type PartialRes2 = PartialTemp<{name: string, age: number}>

  

// Required 去除对象中属性的可选修饰 key-?

  

type RequiredRes = Required<{name: string, age?: number}>

  

type RequiredTemp<T> = {

[K in keyof T]-?: T[K]

}

  

type RequiredRes2 = RequiredTemp<{name: string, age?: number}>

  
  

// Readonly 对象属性都转换为只读 readonly key

  

type ReadonlyRes = Readonly<{name: string, age?: number}>

  

type ReadonlyTemp<T> = {

readonly [K in keyof T]: T[K]

}

  

type ReadonlyRes2 = Readonly<{name: string, age? : number}>

  
  

// Pick 过滤对象属性

  

type PickRes = Pick<{name: string, age?: number}, 'name' | 'age'>

  

type PickTemp<T, K extends keyof T> = {

[P in K]: T[P]

}

  

type PickRes2 = PickTemp<{name: string, age?: number}, 'name' | 'age'>

  
  

// Record 创建索引类型

  

type RecordRes = Record<'a' | 'b', number>

  

type RecordTemp<T extends keyof any, K> = {

[P in T]: K

}

  

type RecordRes2 = RecordTemp<'a' | 'b', number>

type RecordRes3 = RecordTemp<string, any>

  
  

// Exclude 从联合类型中去掉一部分类型

  

type ExcludeRes = Exclude<'a' | 'b' | 'c' | 'd', 'a' | 'b' | 'e'>

  

type ExcludeTemp<T, P> = T extends P ? never : T

  

type ExcludeRes2 = ExcludeTemp<'a' | 'b' | 'c' | 'd', 'a' | 'b' | 'e'>
