// 模式匹配做提取

  

type GetReturnType<Func extends Function> =

Func extends () => infer returnType

? returnType

: never

  
  
  

type str = GetReturnType<() => '123'>

  

// 重新构造做变化

// { name: '123', age: 123 } => {NAME: '123', AGE: 123 }

  

type UppercaseKey<obj extends Record<string, any>> = {

[key in keyof obj as Uppercase<key & string>]: obj[key]

}

  

type obj = UppercaseKey<{ name: '123', age: 123 }>

  

// 递归复用做循环

// 'hello' => h | e | l | o

  

type StringToUnion<str extends string> =

str extends `${infer a}${infer rest}`

? a | StringToUnion<rest>

: never

  

type hello = StringToUnion<'hello'>

  
  
  

// 数组长度做计数

type BuildArray<

Len extends number,

Ele = unknown,

Arr extends unknown[] = []

> = Arr['length'] extends Len

? Arr

: BuildArray<Len, Ele, [...Arr, Ele]>

  

type arr2 = BuildArray<2>

  

type Subtract<Num1 extends number, Num2 extends number> =

BuildArray<Num1> extends [...BuildArray<Num2>, ...infer arr]

? arr['length']

: never

  

type sub = Subtract<32, 11>

str extends `${infer A}&${infer B}`

? Merge<ParseStr<A>, ParseQueryString<B>>

: ParseStr<str>

[K in A]: B

type Merge<obj1 extends Record<string, any>, obj2 extends Record<string, any>> = {

[key in keyof obj1 | keyof obj2]:

key extends keyof obj1

? key extends keyof obj2

? MergeValue<obj1[key], obj2[key]>

: obj1[key]

: key extends keyof obj2

? obj2[key]

: never

}

  
  

type MergeValue<val1, val2> =

val1 extends val2

? val1

: val2 extends unknown[]

? [val1, ...val2]

: [val1, val2]

  
  

type mergeObj = Merge<{a: 1}, {b: 1}>
