// 5. 联合分散可简化

type UppercaseA<str extends string> =

str extends 'a' ? Uppercase<str> : str

  

type UpercaseFirst<str extends string> =

str extends `${infer First}${infer Rest}`

? `${Uppercase<First>}${Rest}`

: str

  

type BEM<Block extends string, Element extends string[], Modifier extends string[]> = `${Block}__${Element[number]}--${Modifier[number]}`

  

type Combination<A extends string, B extends string> =

A | B | `${A}${B}` | `${B}${A}`

  

type AllCombinations<A extends string, B extends string = A> =

A extends A

? Combination<A, AllCombinations<Exclude<B, A>>>

: never;

  
  
  
  

type a = 'a'

type b = 'b'

  
  

type comAB = Combination<a, b>

type comAll = AllCombinations<'a' | 'b'>

  

type union = 'a' | 'b' | 'c'

  

type bc = Exclude<union, 'a'>

  

type str1 = 'normal'

type UppercaseStr1 = UpercaseFirst<str1>

  

type unionA = UppercaseA<union>

  

type str = `${union}++`

  

type bemResult = BEM<'guang', ['aaa', 'bbb'], ['warning', 'success']>;
